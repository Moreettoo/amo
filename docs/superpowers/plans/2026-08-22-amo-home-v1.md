# AMO Home v1 — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Construir a página inicial da AMO — bolos no pote zero açúcar — visualmente completa em 12 seções, com todas as ações comerciais inertes.

**Architecture:** App Router do Next.js servindo uma única rota estática. Conteúdo vem de três arquivos de dados tipados em `src/data/` que espelham 1:1 as futuras tabelas do Supabase. Cada seção é um componente isolado que recebe dados e renderiza; nenhuma seção conhece as outras. Animação concentrada em quatro primitivas em `src/components/ui/`, todas com desligamento por `prefers-reduced-motion`.

**Tech Stack:** Next.js (App Router) · TypeScript strict · Tailwind v4 · Motion (`motion/react`) · next/font/google · Vitest + Testing Library + jsdom + axe-core · Pillow 12.2 (script de asset)

**Spec:** `docs/superpowers/specs/2026-08-22-amo-home-design.md`

`--color-rosewood-forte` **não é enfeite, é correção de acessibilidade medida.**
`#B9707A` reprova em dois usos:

| par | medido | mínimo | veredito |
|---|---|---|---|
| cream sobre `#B9707A` | 3.35:1 | 4.5 (texto normal) | **reprova** |
| `#B9707A` como anel de foco sobre sage | 2.25:1 | 3.0 | **reprova** |
| cream sobre `#9B5E66` | 4.52:1 | 4.5 | passa |
| `#9B5E66` como anel de foco sobre cream / sage / sand / ink | 4.52 / 3.04 / 3.10 / 3.02 | 3.0 | passa nos quatro |

Regra de uso, sem exceção:
- `--color-rosewood` (`#B9707A`) → **só** como superfície da faixa da assinatura, onde o texto é Bodoni caps grande (aí 3.35:1 satisfaz o limite de texto grande, 3.0)
- `--color-rosewood-forte` (`#9B5E66`) → **todo** CTA, **todo** anel de foco, **todo** texto em tamanho normal sobre rosewood

- [ ] **Step 5: Implementar `src/lib/contraste.ts`**

```ts
/** Luminância relativa por WCAG 2.1, formula 
 *  https://www.w3.org/TR/WCAG21/#dfn-relative-luminance */
function luminancia(hex: string): number {
  const v = hex.replace('#', '')
  const canais = [0, 2, 4].map((i) => {
    const c = parseInt(v.slice(i, i + 2), 16) / 255
    return c <= 0.03928 ? c / 12.92 : Math.pow((c + 0.055) / 1.055, 2.4)
  })
  return 0.2126 * canais[0] + 0.7152 * canais[1] + 0.0722 * canais[2]
}

/** Razão de contraste entre duas cores hex. Ordem não importa. */
export function contraste(a: string, b: string): number {
  const la = luminancia(a)
  const lb = luminancia(b)
  const alto = Math.max(la, lb)
  const baixo = Math.min(la, lb)
  return (alto + 0.05) / (baixo + 0.05)
}

/** Extrai os --color-* de um texto CSS. Fonte única de verdade é o CSS. */
export function lerTokens(css: string): Record<string, string> {
  const tokens: Record<string, string> = {}
  const re = /--color-([a-z-]+):\s*(#[0-9A-Fa-f]{6})/g
  let m: RegExpExecArray | null
  while ((m = re.exec(css)) !== null) {
    tokens[m[1]] = m[2].toUpperCase()
  }
  return tokens
}
```

```python
"""Tira o xadrez de transparencia achatado do JPEG do logo AMO.

O arquivo de origem e um JPEG, e JPEG nao guarda canal alpha — o quadriculado
de transparencia foi rasterizado junto com a arte. Medicao do original:
o xadrez e grayscale puro (#FFFFFF e ~#CCCCCC, saturacao 0.000), enquanto o
cobre do logo tem saturacao alta e o contorno tem luminancia baixa. Da para
separar por saturacao + luminancia com rampa suave na borda.

Isto e paliativo. O certo e o logo em SVG ou PNG com alpha exportado do
original — esta na lista de pendencias da spec.
"""
from PIL import Image
import numpy as np
from pathlib import Path

ORIGEM = Path("_referencias/marca/Gemini_Generated_Image_29yuu129yuu129yu.jpeg")
DESTINO = Path("public/marca")

# Abaixo de SAT_MIN a cor e considerada cinza (candidata a xadrez).
# Acima de VAL_MIN ela e clara demais para ser o contorno preto da arte.
SAT_MIN = 0.12
VAL_MIN = 0.72
# Largura da rampa de alpha, para a borda anti-aliasada nao serrilhar.
RAMPA = 0.10


def recortar(img: Image.Image) -> Image.Image:
    rgb = np.asarray(img.convert("RGB"), dtype=np.float32) / 255.0
    mx = rgb.max(axis=2)
    mn = rgb.min(axis=2)
    val = mx
    sat = np.where(mx > 0, (mx - mn) / np.maximum(mx, 1e-6), 0.0)

    # 1.0 = totalmente opaco, 0.0 = totalmente transparente.
    # Pixel cinza E claro -> xadrez -> transparente.
    e_cinza = np.clip((SAT_MIN - sat) / SAT_MIN, 0.0, 1.0)
    e_claro = np.clip((val - VAL_MIN) / RAMPA, 0.0, 1.0)
    alpha = 1.0 - (e_cinza * e_claro)

    saida = np.dstack([
        (rgb * 255).astype(np.uint8),
        (alpha * 255).astype(np.uint8),
    ])
    return Image.fromarray(saida, mode="RGBA")


def recortar_caixa(img: Image.Image, box: tuple) -> Image.Image:
    """Corta e aparа a moldura transparente que sobrar."""
    corte = img.crop(box)
    limites = corte.getbbox()
    return corte.crop(limites) if limites else corte


def main() -> None:
    if not ORIGEM.exists():
        raise SystemExit(f"origem nao encontrada: {ORIGEM}")

    DESTINO.mkdir(parents=True, exist_ok=True)
    original = Image.open(ORIGEM)
    L, A = original.size
    print(f"origem: {L}x{A}")

    limpo = recortar(original)

    # Lockup completo: a arte inteira, para o footer.
    completo = recortar_caixa(limpo, (0, 0, L, A))
    completo.save(DESTINO / "logo-amo-lockup.png", optimize=True)
    print(f"logo-amo-lockup.png: {completo.size}")

    # Lockup curto: so "A m [pote]", a faixa de cima. Para a nav, onde o
    # descritor e a assinatura ficariam ilegiveis a 40px de altura.
    curto = recortar_caixa(limpo, (0, 0, L, int(A * 0.46)))
    curto.save(DESTINO / "logo-amo.png", optimize=True)
    print(f"logo-amo.png: {curto.size}")


if __name__ == "__main__":
    main()
```

```tsx
  delay?: number

export function Reveal({ children, delay = 0, className }: Props) {
  const reduzir = useReducedMotion()

  // Sob prefers-reduced-motion não há transição nenhuma: o conteúdo
  // simplesmente já está no estado final.
  if (reduzir) return <div className={className}>{children}</div>

  return (
    <motion.div
      className={className}
      initial={{ opacity: 0, y: 16 }}
      whileInView={{ opacity: 1, y: 0 }}
      viewport={{ once: true, margin: '-10%' }}
      transition={{ duration: 0.5, delay, ease: EASE_AMO }}
    >
      {children}
```

```tsx
'use client'

import Image from 'next/image'
import Link from 'next/link'
import { useEffect, useState } from 'react'

const ITENS = [
  { href: '#sabores', rotulo: 'Sabores' },
  { href: '#marca', rotulo: 'A marca' },
  { href: '#ritual', rotulo: 'O ritual' },
] as const

export function Nav() {
  const [rolou, setRolou] = useState(false)
  const [aberto, setAberto] = useState(false)

  useEffect(() => {
    const aoRolar = () => setRolou(window.scrollY > 40)
    aoRolar()
    window.addEventListener('scroll', aoRolar, { passive: true })
    return () => window.removeEventListener('scroll', aoRolar)
  }, [])

  useEffect(() => {
    if (!aberto) return
    const aoTeclar = (e: KeyboardEvent) => {
      if (e.key === 'Escape') setAberto(false)
    }
    window.addEventListener('keydown', aoTeclar)
    return () => window.removeEventListener('keydown', aoTeclar)
  }, [aberto])

  return (
    <div className="sticky top-3 z-40 px-3 md:top-4 md:px-6">
      <nav
        aria-label="Principal"
        className={
          'mx-auto flex h-[88px] max-w-[1280px] items-center justify-between ' +
          'rounded-[20px] px-5 md:px-8 ' +
          'transition-[background-color,box-shadow] duration-[240ms] ' +
          (rolou
            ? 'bg-sage/95 shadow-[0_8px_32px_-8px_rgb(46_36_31_/_0.12)] backdrop-blur-md'
            : 'bg-transparent')
        }
      >
        <Link href="#conteudo" className="shrink-0">

            src="/marca/logo-amo.png"
            alt="AMO Sabores Saudáveis"
```

```tsx
'use client'

import Image from 'next/image'
import { motion, useReducedMotion } from 'motion/react'
import { Etiqueta } from '@/components/ui/Etiqueta'
import { Pill } from '@/components/ui/Pill'
import { Estrato } from '@/components/ui/Estrato'
import { EASE_AMO } from '@/lib/motion'
import { SABORES } from '@/data/sabores'

/**
 * As três linhas do título, cada uma em sua faixa de tonalidade.
 * É a tese da página: um bolo no pote é uma secção transversal, e o
 * título é essa secção. Ver spec §4.
 */
const LINHAS = [
  { texto: 'Um abraço em', fundo: 'bg-sage' },
  { texto: 'forma de', fundo: 'bg-sage/80' },
  { texto: 'sobremesa', fundo: 'bg-sage/60' },
] as const

const POTE = SABORES[0]

export function Hero() {
  const reduzir = useReducedMotion()

  // Sob reduced-motion nada anima: tudo já nasce no estado final.
  const anim = (delay: number) =>
    reduzir
      ? {}
      : {
          initial: { opacity: 0, y: 20 },
          animate: { opacity: 1, y: 0 },
          transition: { duration: 0.6, delay, ease: EASE_AMO },
        }

  const animFaixa = (delay: number) =>
    reduzir
      ? {}
      : {
          initial: { scaleY: 0, opacity: 0 },
          animate: { scaleY: 1, opacity: 1 },
          transition: { duration: 0.5, delay, ease: EASE_AMO },
        }

  return (
    <section aria-labelledby="hero-titulo" className="relative bg-sage pt-8 md:pt-4">
      <div className="mx-auto grid max-w-[1280px] items-center gap-10 px-5 pb-20 md:grid-cols-[1.15fr_1fr] md:gap-14 md:px-10 md:pb-28">
        <div>
          <motion.div {...anim(0.95)}>
            <Etiqueta>Zero açúcar · Cozido a vapor</Etiqueta>
          </motion.div>

          <h1
            id="hero-titulo"
            className="mt-7 flex flex-col text-[clamp(2.75rem,7.5vw,6.5rem)] font-normal leading-[0.94] tracking-[-0.02em] text-ink"
          >
            {LINHAS.map((l, i) => (
              <span key={l.texto} className="relative">
                <motion.span
                  aria-hidden="true"
                  data-faixa
                  {...animFaixa(0.05 + i * 0.09)}
                  className={`absolute inset-x-0 inset-y-[-0.06em] -z-10 origin-top ${l.fundo}`}
                />
                <motion.span
                  className="block"
                  {...anim(0.35 + i * 0.11)}
                >
                  {l.texto}
                </motion.span>
              </span>
            ))}
          </h1>

          <motion.p
            {...anim(0.95)}
            className="mt-7 font-script text-[clamp(1.75rem,3vw,2.5rem)] leading-tight text-rosewood-forte"
          >
            Para quem ama comer bem.
          </motion.p>

          <motion.p
            {...anim(0.95)}
            className="mt-5 max-w-[46ch] font-sans text-[1.0625rem] leading-[1.65] text-ink/75"
          >
            Sete sabores de bolo no pote, montados à mão e adoçados com eritritol
            e&nbsp;stevia. Em pote de&nbsp;vidro que fica com você.
          </motion.p>

          <motion.div
            {...anim(0.95)}
```

```tsx
'use client'

import Image from 'next/image'
import { useEffect, useRef, useState } from 'react'
import { SABORES } from '@/data/sabores'
import { Estrato } from '@/components/ui/Estrato'

const ID_PAINEL = 'preview-sabor'
const idAba = (slug: string) => `aba-${slug}`

export function ListaPinada() {
  const [ativo, setAtivo] = useState(0)
  const refSecao = useRef<HTMLElement | null>(null)
  const refsAba = useRef<(HTMLButtonElement | null)[]>([])

  // O scroll dirige exatamente o mesmo estado que o clique. Gesto nunca é o
  // único caminho — é só um caminho a mais.
  useEffect(() => {
    const secao = refSecao.current
    if (!secao) return

    const aoRolar = () => {
      const r = secao.getBoundingClientRect()
      const percorrido = -r.top
      const total = r.height - window.innerHeight
      if (total <= 0) return
      const p = Math.min(Math.max(percorrido / total, 0), 0.999)
      setAtivo(Math.floor(p * SABORES.length))
    }

    aoRolar()
    window.addEventListener('scroll', aoRolar, { passive: true })
    return () => window.removeEventListener('scroll', aoRolar)
  }, [])

  const mover = (delta: number) => {
    const proximo = (ativo + delta + SABORES.length) % SABORES.length
    setAtivo(proximo)
    refsAba.current[proximo]?.focus()
  }

  const aoTeclar = (e: React.KeyboardEvent) => {
    if (e.key === 'ArrowDown' || e.key === 'ArrowRight') {
      e.preventDefault()
      mover(1)
    } else if (e.key === 'ArrowUp' || e.key === 'ArrowLeft') {
      e.preventDefault()
      mover(-1)
    } else if (e.key === 'Home') {
      e.preventDefault()
      setAtivo(0)
      refsAba.current[0]?.focus()
    } else if (e.key === 'End') {
      e.preventDefault()
      const ultimo = SABORES.length - 1
      setAtivo(ultimo)
      refsAba.current[ultimo]?.focus()
    }
  }

  const sabor = SABORES[ativo]

  return (
    <section
      ref={refSecao}
      aria-labelledby="lista-titulo"
      className="relative bg-cream"
      style={{ height: `${SABORES.length * 55}vh` }}
    >
      <div className="sticky top-0 flex min-h-screen items-center">
        <div className="mx-auto grid w-full max-w-[1280px] items-center gap-10 px-5 py-16 md:grid-cols-[0.85fr_1fr] md:gap-16 md:px-10">
          <div
            id={ID_PAINEL}
            role="tabpanel"
            aria-labelledby={idAba(sabor.slug)}
            className="order-2 overflow-hidden rounded-[24px] md:order-1"
          >
            <div className="aspect-[3/4] w-full">
              <Image
                key={sabor.slug}
                src={sabor.imagem}
                alt={sabor.alt}
                width={sabor.largura}
                height={sabor.altura}
                loading="lazy"
                sizes="(max-width: 768px) 90vw, 40vw"
                className="h-full w-full object-cover motion-safe:animate-[fadeIn_300ms_ease-out]"
              />
            </div>
          </div>

          <div className="order-1 md:order-2">
            <h2
              id="lista-titulo"
              className="font-sans text-[0.6875rem] font-medium uppercase tracking-[0.18em] text-ink/55"
            >
              Todos os sabores
            </h2>

            <div
              role="tablist"
              aria-orientation="vertical"
              aria-labelledby="lista-titulo"
              onKeyDown={aoTeclar}
              className="mt-7 flex flex-col items-start gap-1"
            >
              {SABORES.map((s, i) => {
                const selecionado = i === ativo
                return (
                  <button
                    key={s.slug}
                    id={idAba(s.slug)}
                    ref={(el) => {
                      refsAba.current[i] = el
                    }}
                    role="tab"
                    type="button"
                    aria-selected={selecionado}
                    aria-controls={ID_PAINEL}
                    tabIndex={selecionado ? 0 : -1}
                    onClick={() => setAtivo(i)}
                    className={
                      'rounded-lg text-left font-display ' +
                      'text-[clamp(2rem,5vw,4.5rem)] leading-[1.02] tracking-[-0.02em] ' +
                      'transition-[color,opacity] duration-300 ' +
                      'focus-visible:outline-2 focus-visible:outline-offset-4 ' +
                      'focus-visible:outline-rosewood-forte ' +
                      (selecionado ? 'text-ink' : 'text-ink/25 hover:text-ink/50')
                    }
                  >
                    <span translate="no">{s.nome}</span>
                  </button>
                )
              })}
            </div>

            <p className="mt-7 min-h-[3rem] max-w-[36ch] font-sans text-[0.9375rem] leading-[1.6] text-ink/70">
              {sabor.sabor} — {sabor.frase}
            </p>
          </div>
        </div>
      </div>

      <Estrato
        n={4}
        corDestino="fill-ink"
        className="absolute bottom-0 left-0 -mb-px"
      />
    </section>
  )
}
```

```tsx
import { SELOS } from '@/data/selos'
import { RITUAL } from '@/data/ritual'

describe('a página completa', () => {
  it('tem exatamente um h1', () => {
    render(<Home />)
    expect(screen.getAllByRole('heading', { level: 1 })).toHaveLength(1)
  })

  it('não pula nível de heading', () => {
    const { container } = render(<Home />)
    const niveis = [...container.querySelectorAll('h1,h2,h3,h4,h5,h6')].map((h) =>
      Number(h.tagName[1]),
    )
    for (let i = 1; i < niveis.length; i++) {
      expect(niveis[i] - niveis[i - 1]).toBeLessThanOrEqual(1)
    }
  })

  it('mostra os 7 sabores, os 6 selos e os 4 passos', () => {
    render(<Home />)
    for (const s of SABORES) expect(screen.getAllByText(s.nome).length).toBeGreaterThan(0)
    for (const s of SELOS) expect(screen.getAllByText(s.nome).length).toBeGreaterThan(0)
    for (const p of RITUAL) expect(screen.getAllByText(p.titulo).length).toBeGreaterThan(0)
  })

  it('não deixa imagem sem alt', () => {
    const { container } = render(<Home />)
    for (const img of container.querySelectorAll('img')) {
      expect(img.hasAttribute('alt')).toBe(true)
    }
  })

  it('não deixa imagem sem width e height', () => {
    const { container } = render(<Home />)
    for (const img of container.querySelectorAll('img')) {
      expect(img.hasAttribute('width')).toBe(true)
      expect(img.hasAttribute('height')).toBe(true)
    }
  })

  it('todo link interno aponta para um id que existe na página', () => {
    const { container } = render(<Home />)
    const ancoras = [...container.querySelectorAll('a[href^="#"]')]
      .map((a) => a.getAttribute('href')!.slice(1))
      .filter((id) => id && id !== 'conteudo')
    for (const id of new Set(ancoras)) {
      expect(container.querySelector(`#${id}`), `âncora morta: #${id}`).toBeTruthy()
    }
  })

  it('não tem afordância morta: nenhum botão sem handler nem link vazio', () => {
    const { container } = render(<Home />)
    for (const a of container.querySelectorAll('a')) {
      expect(a.getAttribute('href')).toBeTruthy()
    }
  })

  it('não mostra preço em lugar nenhum — está fora do escopo da v1', () => {
    const { container } = render(<Home />)
    expect(container.textContent).not.toMatch(/R\$\s?\d/)
  })

  it('não usa reticências de três pontos', () => {
    const { container } = render(<Home />)
    expect(container.textContent).not.toMatch(/\.\.\./)
  })

  it('passa no axe sem violação', async () => {
    const { container } = render(<Home />)
    expect(await axe(container)).toHaveNoViolations()
  })
})
```
